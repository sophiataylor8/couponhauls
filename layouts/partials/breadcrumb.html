{{/* breadcrumb.html Renders HTML breadcrumb nav AND JSON-LD BreadcrumbList schema. Works correctly for store pages (subdomain URLs) and all other pages. */}} {{- $siteURL := .Site.BaseURL | strings.TrimRight "/" -}} {{- $section := .Section -}} {{- $items := slice (dict "name" "Home" "url" (printf "%s/" $siteURL)) -}} {{- if eq $section "blog" -}} {{- $items = $items | append (dict "name" "Blog" "url" (printf "%s/blog/" $siteURL)) -}} {{- $items = $items | append (dict "name" .Title "url" .Permalink) -}} {{- else if eq $section "stores" -}} {{- $storeName := .Params.store_name | default .Title | replaceRE `^"(.*)"$` "$1" | strings.TrimSpace -}} {{- $pageURL := partial "canonical-url.html" . -}} {{- $items = $items | append (dict "name" "Stores" "url" (printf "%s/stores/" $siteURL)) -}} {{- $items = $items | append (dict "name" (printf "%s Coupon Codes" $storeName) "url" $pageURL) -}} {{- else -}} {{- $items = $items | append (dict "name" .Title "url" .Permalink) -}} {{- end -}} {{/* ── HTML breadcrumb ─────────────────────────────────────────────── */}}
    {{ range $i, $item := $items }} {{ $isLast := eq $i (sub (len $items) 1) }}
  1. {{ if not $isLast }} {{ $item.name }} {{ else }} {{ $item.name }} {{ end }}
    2. {{ end }}
{{/* ── JSON-LD BreadcrumbList ───────────────────────────────────────── */}}
