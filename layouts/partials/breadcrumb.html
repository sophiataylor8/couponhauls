{{/* layouts/partials/breadcrumb.html Renders an HTML breadcrumb nav AND a JSON-LD BreadcrumbList. Works for store pages (subdomain URLs), blog pages and all other pages. Fixes applied: - Every "| jsonify" inside the ld+json script is followed by "| safeJS" (without it Hugo escapes the already-quoted JSON string a second time, producing URLs like /%22https://...%22 that Google reports as 404s) - Duplicate schema.org microdata removed from the HTML — JSON-LD is the single source of truth, so Search Console no longer sees two BreadcrumbLists on one page - Main-domain URLs (Home / Blog / Stores) are built from .Site.BaseURL */}} {{- $siteURL := .Site.BaseURL | strings.TrimRight "/" -}} {{- $section := .Section -}} {{- $items := slice (dict "name" "Home" "url" (printf "%s/" $siteURL)) -}} {{- if eq $section "blog" -}} {{- $items = $items | append (dict "name" "Blog" "url" (printf "%s/blog/" $siteURL)) -}} {{- $items = $items | append (dict "name" .Title "url" .Permalink) -}} {{- else if eq $section "stores" -}} {{- $storeName := .Params.store_name | default .Title | replaceRE `^"(.*)"$` "$1" | strings.TrimSpace -}} {{- $pageURL := partial "canonical-url.html" . -}} {{- $items = $items | append (dict "name" "Stores" "url" (printf "%s/stores/" $siteURL)) -}} {{- $items = $items | append (dict "name" (printf "%s Coupon Codes" $storeName) "url" $pageURL) -}} {{- else -}} {{- $items = $items | append (dict "name" .Title "url" .Permalink) -}} {{- end -}} {{- $lastIdx := sub (len $items) 1 -}} {{/* ── HTML breadcrumb ─────────────────────────────────────────────── */}}
    {{- range $i, $item := $items }}
  1. {{- if lt $i $lastIdx }} {{ $item.name }} {{- else }} {{ $item.name }} {{- end }}
    2. {{- end }}
{{/* ── JSON-LD BreadcrumbList ──────────────────────────────────────── */}}
